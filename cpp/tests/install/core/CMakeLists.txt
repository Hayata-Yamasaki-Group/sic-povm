cmake_minimum_required(VERSION 3.20)
project(core-consumer LANGUAGES CXX)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
# Exercise the native half of installed GCC fat LTO archives explicitly.
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  add_compile_options(-fno-lto)
  add_link_options(-fno-lto)
endif()
find_package(sic REQUIRED COMPONENTS core support search)
get_target_property(defs sic::core INTERFACE_COMPILE_DEFINITIONS)
if(defs MATCHES "SIC_HAVE_FLINT|SOURCE_ID")
  message(FATAL_ERROR "core exports private build definitions: ${defs}")
endif()
get_target_property(links sic::core INTERFACE_LINK_LIBRARIES)
if(links MATCHES "FLINT|sic_exact")
  message(FATAL_ERROR "binary64 core exports exact dependencies: ${links}")
endif()
add_executable(core-consumer main.cpp)
target_link_libraries(core-consumer PRIVATE sic::core sic::support sic::search)
