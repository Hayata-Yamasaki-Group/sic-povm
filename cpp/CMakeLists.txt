if(SIC_SOURCE_COMMIT STREQUAL "")
  if(EXISTS "${PROJECT_SOURCE_DIR}/SOURCE_COMMIT")
    file(READ "${PROJECT_SOURCE_DIR}/SOURCE_COMMIT" SIC_SOURCE_COMMIT)
    string(STRIP "${SIC_SOURCE_COMMIT}" SIC_SOURCE_COMMIT)
  else()
    execute_process(COMMAND git -C "${PROJECT_SOURCE_DIR}" rev-parse HEAD
      OUTPUT_VARIABLE SIC_SOURCE_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  endif()
endif()

function(sic_map_dependency target)
  foreach(_root IN LISTS ARGN)
    if(IS_ABSOLUTE "${_root}" AND IS_DIRECTORY "${_root}")
      string(REGEX REPLACE "/+$" "" _root "${_root}")
      set_property(TARGET ${target} APPEND PROPERTY INTERFACE_COMPILE_OPTIONS
        "$<BUILD_INTERFACE:-ffile-prefix-map=${_root}/=/sic-dependency/>")
      set_property(GLOBAL APPEND PROPERTY SIC_DEPENDENCY_PATHS "${_root}")
      if(_root MATCHES "^(.+)/include(/.*)?$")
        set(_prefix "${CMAKE_MATCH_1}")
        if(NOT _prefix STREQUAL "/usr" AND NOT _prefix STREQUAL "/usr/local")
          set_property(GLOBAL APPEND PROPERTY SIC_DEPENDENCY_PATHS "${_prefix}")
        endif()
      endif()
    endif()
  endforeach()
endfunction()
foreach(_prefix IN LISTS CMAKE_PREFIX_PATH)
  string(REGEX REPLACE "/+$" "" _prefix "${_prefix}")
  if(IS_ABSOLUTE "${_prefix}" AND NOT _prefix STREQUAL "/usr" AND NOT _prefix STREQUAL "/usr/local")
    set_property(GLOBAL APPEND PROPERTY SIC_DEPENDENCY_PATHS "${_prefix}")
  endif()
endforeach()
